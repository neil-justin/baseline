### Task 2: Export script + PNG/ICO outputs

**Files:**
- Create: `scripts/generate-icons.mjs`
- Create (generated, committed): `app/apple-icon.png`, `app/favicon.ico`

- [ ] **Step 1: Create the script**

```js
/**
 * Renders app/icon.svg into app/apple-icon.png (180×180, iOS home screen)
 * and app/favicon.ico (32×32, legacy browser/crawler fallback).
 * Re-run after editing the source SVG: node scripts/generate-icons.mjs
 *
 * sharp is already in node_modules as a Next.js transitive dependency — do
 * NOT add it to package.json without user sign-off (spec 2026-11-04).
 * ICO output is a minimal container embedding a PNG (Vista+ format):
 * 6-byte header + one 16-byte directory entry + the PNG blob.
 */
import { readFile, writeFile } from "node:fs/promises";
import sharp from "sharp";

function icoFromPng(png) {
  const header = Buffer.alloc(6);
  header.writeUInt16LE(0, 0); // reserved
  header.writeUInt16LE(1, 2); // 1 = icon file
  header.writeUInt16LE(1, 4); // image count
  const entry = Buffer.alloc(16);
  entry.writeUInt8(32, 0); // width in px (32; 0 would mean 256)
  entry.writeUInt8(32, 1); // height in px
  entry.writeUInt8(0, 2); // no local palette
  entry.writeUInt8(0, 3); // reserved
  entry.writeUInt16LE(1, 4); // color planes
  entry.writeUInt16LE(32, 6); // bits per pixel
  entry.writeUInt32LE(png.length, 8); // data size
  entry.writeUInt32LE(22, 12); // data offset = 6 + 16
  return Buffer.concat([header, entry, png]);
}

const svg = await readFile(new URL("../app/icon.svg", import.meta.url));

// density up-scales rasterization of the 64px viewBox so 180px stays crisp
await sharp(svg, { density: 300 })
  .resize(180, 180)
  .png()
  .toFile(new URL("../app/apple-icon.png", import.meta.url));

const png32 = await sharp(svg, { density: 300 }).resize(32, 32).png().toBuffer();
await writeFile(new URL("../app/favicon.ico", import.meta.url), icoFromPng(png32));

console.log("wrote app/apple-icon.png and app/favicon.ico");
```

```

/**
 * The Baseline brand mark — the favicon geometry as an inline SVG React
 * component (spec docs/superpowers/specs/2026-11-04-app-icon-design.md).
 * Purely decorative: it always sits beside the "Baseline" wordmark, so it
 * is aria-hidden. The browser-tab icon lives in app/icon.svg; keep the two
 * in sync if the mark ever changes.
 */
export default function BrandMark({ size = 20 }: { size?: number }) {
  return (
    <svg
      width={size}
      height={size}
      viewBox="0 0 64 64"
      aria-hidden="true"
      focusable="false"
      style={{ display: "block", flex: "none" }}
    >
      <rect x="4" y="4" width="56" height="56" rx="14" fill="#012B53" />
      <path
        fill="#FFFFFF"
        fillRule="evenodd"
        d="M20 13 H35 Q44 13 44 20.5 Q44 25.5 40 27 Q45.5 28.5 45.5 35 Q45.5 43 35.5 43 H20 Z M27 19.5 V25.5 H34 Q37.5 25.5 37.5 22.5 Q37.5 19.5 34 19.5 Z M27 30.5 V36.5 H35 Q39 36.5 39 33.5 Q39 30.5 35 30.5 Z"
      />
      <rect x="20" y="47.5" width="24" height="5" rx="2.5" fill="#1A73E8" />
    </svg>
  );
}
```
